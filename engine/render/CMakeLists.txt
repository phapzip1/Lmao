# Renderer building blocks on the RHI. Only the pipeline library so far.
add_library(lmao_render STATIC
    pipeline_library.h
    pipeline_library.cpp
)
add_library(Lmao::render ALIAS lmao_render)

target_link_libraries(lmao_render PUBLIC Lmao::core Lmao::rhi)

# What PipelineLibrary hot reload runs: the same slangc and flags as lmao_add_shaders. Only
# included when LMAO_SHADER_HOT_RELOAD is on, so Release binaries don't embed these paths.
lmao_find_slangc(slangc)
list(TRANSFORM LMAO_SLANGC_FLAGS PREPEND "\"" OUTPUT_VARIABLE quoted_flags)
list(TRANSFORM quoted_flags APPEND "\"")
list(JOIN quoted_flags ", " LMAO_SLANGC_FLAGS_CPP)
set(LMAO_SLANGC_PATH "${slangc}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/render/slangc_config.h" CONTENT [[
// Generated by engine/render/CMakeLists.txt. Do not edit.
#pragma once

#include <string_view>

namespace Lmao::Render::Generated
{
    inline constexpr std::string_view SlangcPath = "@LMAO_SLANGC_PATH@";
    inline constexpr std::string_view SlangcFlags[] = {@LMAO_SLANGC_FLAGS_CPP@};
    inline constexpr std::string_view ShaderIncludeDirectory = "@LMAO_SHADER_INCLUDE_DIR@";
} // namespace Lmao::Render::Generated
]] @ONLY)
target_include_directories(lmao_render PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

lmao_set_warnings(lmao_render)
set_target_properties(lmao_render PROPERTIES FOLDER "engine")

add_subdirectory(imgui)
